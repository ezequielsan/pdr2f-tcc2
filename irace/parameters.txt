## parameters.txt: espaco de busca do irace para o BRKGA-MP-IPR (PDR2F)
##
## Oito parametros, conforme claude/protocolo_experimental_fechado.md. Os demais
## ficam fixos em base.conf. A calibracao e' feita no BRKGA-MP-IPR; o BRKGA-MP
## herda os mesmos valores, com ipr_interval = 0, para que a diferenca entre os
## dois bracos meca apenas o efeito do IPR.
##
## O "switch" (segunda coluna) e' o nome da chave no arquivo de configuracao da
## API: o target-runner substitui essas chaves em base.conf.
##
## nome                         switch                          tipo   faixa

population_size                "population_size "              i,log  (50, 1000)
elite_percentage               "elite_percentage "             r      (0.10, 0.50)
mutants_percentage             "mutants_percentage "           r      (0.01, 0.30)
num_elite_parents              "num_elite_parents "            i      (1, 5)
total_parents                  "total_parents "                i      (2, 8)
bias_type                      "bias_type "                    c      (CONSTANT, LINEAR, QUADRATIC, CUBIC, EXPONENTIAL, LOGINVERSE)
num_independent_populations    "num_independent_populations "  i      (1, 3)
ipr_interval                   "ipr_interval "                 i      (10, 200)

## --- Restricoes ------------------------------------------------------------
## No irace 4 as restricoes ficam nesta secao (o antigo forbiddenFile do
## cenario foi removido). Uma expressao logica do R por linha; configuracoes
## que a satisfazem nunca sao amostradas.
##
## A API exige num_elite_parents < total_parents (estrito) e que o conjunto
## elite tenha pelo menos num_elite_parents individuos. Com as faixas acima a
## segunda condicao nunca e' violada (elite >= 0,10 * 50 = 5), mas fica
## registrada por seguranca.
[forbidden]
num_elite_parents >= total_parents
floor(elite_percentage * population_size) < num_elite_parents

## --- Opcoes globais --------------------------------------------------------
## Casas decimais dos parametros reais (antes era a opcao digits do cenario).
[global]
digits = 4
